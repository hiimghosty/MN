def dy(x, y):
    return 2*x*y

def d2y(x, y):
    # (dy)'_x + (dy)'_y * dy
    return grad(dy, argnums=0)(x, y) + grad(dy, argnums=1)(x, y) * dy(x, y)

    # (dy)'_x + (dy)'_y * dy
    return grad(d2y, argnums=0)(x, y) + grad(d2y, argnums=1)(x, y) * dy(x, y)
