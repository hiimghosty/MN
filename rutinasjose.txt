################RUTINA010-Coef. de correlacion##########################
#
#Qué hace: Halla el coeficiente de correlación de Pearson.
#Qué necesita: Valores de predicción de las “y” que devuelve la función hallada por
medio de mínimos cuadrados al evaluar las “x” de dato tabulados, la media de los valores
de “y” de dato tabulados y la media de los valores de predicción de “y”.
#rutina:
def R2(y_data, y_pred):
    y_mean = np.mean(y_data)
    SS_res = np.sum((y_data - y_pred) ** 2)
    SS_tot = np.sum((y_data - y_mean) ** 2)
    return 1 - SS_res / SS_tot
